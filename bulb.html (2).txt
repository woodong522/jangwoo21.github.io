    <script>
        function light(){
            let val = Number(document.getElementById("numIn").value);
            let lamps = document.querySelectorAll(".lamp");
            // 先全部熄灭
            for(let i=0;i<8;i++){
                lamps[i].classList.remove("on");
            }
            // 按二进制位点亮
            for(let i=0;i<8;i++){
                if(val & (1 << i)){
                    lamps[i].classList.add("on");
                }
            }
        }
    </script>
